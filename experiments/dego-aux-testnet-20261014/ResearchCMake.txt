cmake_minimum_required(VERSION 3.16)
project(DeroGold VERSION 1.0.1.0 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -pthread")
add_compile_options(-maes -msse2)
add_compile_definitions(DEGO_RESEARCH_TESTNET RAPIDJSON_HAS_STDSTRING _DEFAULT_SOURCE _GNU_SOURCE FORCE_USE_HEAP)
set(BUILD_COMMIT_ID "aux-testnet-20261014-NOT-MAINNET")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/version")
configure_file(src/config/version.h.in "${CMAKE_BINARY_DIR}/version/version.h")
add_custom_target(version ALL)
include_directories("${CMAKE_BINARY_DIR}/version" external include src src/platform/linux src/platform/posix)
find_package(Boost REQUIRED COMPONENTS serialization)
find_package(OpenSSL REQUIRED)
find_package(cxxopts REQUIRED)
add_library(rapidjson INTERFACE)
add_library(httplib::httplib INTERFACE IMPORTED)
foreach(pair "cryptopp|cryptopp" "miniupnpc|miniupnpc")
 string(REPLACE "|" ";" parts "${pair}")
 list(GET parts 0 name)
 list(GET parts 1 lib)
 add_library(${name}::${name} INTERFACE IMPORTED)
 set_property(TARGET ${name}::${name} PROPERTY INTERFACE_LINK_LIBRARIES ${lib})
endforeach()
add_library(RocksDB::rocksdb INTERFACE IMPORTED)
set_property(TARGET RocksDB::rocksdb PROPERTY INTERFACE_LINK_LIBRARIES "rocksdb")
add_subdirectory(src)
target_compile_options(cryptotest PRIVATE -UNDEBUG)
add_executable(research-inspect research_inspect.cpp)
target_include_directories(research-inspect PRIVATE external/nlohmann-json)
target_link_libraries(research-inspect CryptoNoteCore)
