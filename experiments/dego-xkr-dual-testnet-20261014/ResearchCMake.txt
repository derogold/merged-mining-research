cmake_minimum_required(VERSION 3.16)
project(Kryptokrona LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -pthread")
add_compile_options(-maes -msse2)
add_compile_definitions(USE_TESTNET RAPIDJSON_HAS_STDSTRING _DEFAULT_SOURCE _GNU_SOURCE FORCE_USE_HEAP)
set(VERSION "research-cf5a63e-NOT-MAINNET")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/version")
configure_file(src/config/version.h.in "${CMAKE_BINARY_DIR}/version/version.h")
add_custom_target(version ALL)
include_directories("${CMAKE_BINARY_DIR}/version" external include src src/platform/linux src/platform/posix)
find_package(Boost REQUIRED COMPONENTS system filesystem thread date_time chrono serialization program_options)
foreach(pair "rocksdblib|rocksdb" "upnpc-static|miniupnpc" "cryptopp-static|cryptopp")
 string(REPLACE "|" ";" parts "${pair}")
 list(GET parts 0 name)
 list(GET parts 1 lib)
 add_library(${name} INTERFACE)
 target_link_libraries(${name} INTERFACE ${lib})
endforeach()
add_subdirectory(src)
target_compile_options(crypto_test PRIVATE -UNDEBUG)
set_source_files_properties("${CMAKE_SOURCE_DIR}/src/cryptonote_core/rocksdb_wrapper.cpp" DIRECTORY src PROPERTIES COMPILE_OPTIONS "-fno-rtti")
add_executable(xkr-template xkr-template.cpp)
target_include_directories(xkr-template PRIVATE external/nlohmann-json)
target_link_libraries(xkr-template cryptonote_core)
target_link_libraries(xkr-template rocksdblib)
target_link_libraries(cryptonote_core rocksdblib)
# Match the earlier known-answer-tested hashing build; GCC O3 failed a variant-1 vector.
target_compile_options(crypto PRIVATE -O2)
